######## Full model for SN ################

par ATOT=1
par KTOT=500
par PTOT=107.5

par k1=0.02,k11=1
par k2=0.01
par k3=0.032,k33=1
par k4=15
par k5=0.05,k55=1
par k6=0.01
par k7=1,k77=0.01
par k8=0.15


########### Conservation relationship ##############################
A = ATOT-(AP + APP + AK + APK + APPC + APC)
K= KTOT-(AK + APK)
P= PTOT-(APPC + APC)

#################### rates #########################################


## R1: A + K <---k1/k11--> AK
## R2: AK ---k2---> AP + K
## R3: AP + K <---k3/k33--> APK
## R4: APK --k4---> APP + K
## R5: APP + P <---k5/k55---> APPC
## R6: APPC --- k6----> AP + P
## R7: AP <---- k7/k77---> APC 
## R8: APC ---k8 ---> A + P


v1=k1*A*K-k11*AK
v2=k2*AK
v3=k3*AP*K-k33*APK
v4=k4*APK 
v5=k5*APP*P-k55*APPC
v6=k6*APPC
v7=k7*AP*P-k77*APC
v8=k8*APC

############# ODE ###########################

AP'=v2-v3+v6-v7
APP'=v4-v5
AK'=v1-v2
APK'=v3-v4
APPC'=v5-v6
APC'=v7-v8
