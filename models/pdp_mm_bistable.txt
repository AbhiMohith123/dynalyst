########### Conservation relationship ##############################

AP = ATOT-(A + APP)

#################### rates #########################################


## R1: A + K <---k1/k11--> AK
## R2: AK ---k2---> AP + K
## R3: AP + K <---k3/k33--> APK
## R4: APK --k4---> APP + K
## R5: APP + P <---k5/k55---> APPC
## R6: APPC --- k6----> AP + P
## R7: AP <---- k7/k77---> APC 
## R8: APC ---k8 ---> A + P


r1 = (k2*KTOT*A/km1)/(1 + A/km1 + (ATOT-A-APP)/km3)
r2 = (k4*KTOT*(ATOT-A-APP)/km3)/(1+ A/km1 + (ATOT-A-APP)/km3)

r3 = (k8*PTOT*(ATOT-A-APP)/km7)/(1+  (ATOT-A-APP)/km7+ APP/km5)
r4 = (k6*PTOT*APP/km5)/(1+ (ATOT-A-APP)/km7+ APP/km5)

############# ODE ###########################

  A'=r3-r1
APP'=r2-r4
